/** * Honeymoon fund backend (Google Apps Script + Stripe Checkout) * * SETUP (once) * 1. In the Google Sheet: Extensions > Apps Script. Delete the default code and paste this whole file. * 2. Project Settings (gear icon) > Script properties > Add property: * Name: STRIPE_SECRET_KEY * Value: your Stripe secret key (sk_test_... while testing, sk_live_... for real money) * 3. In the function dropdown, choose `setup` and click Run. Approve the permissions. * 4. Deploy > New deployment > type: Web app. Execute as: Me. Who has access: Anyone. Deploy. * Copy the Web app URL (ends in /exec). * 5. Paste that URL into CHECKOUT_ENDPOINT in the website file and in thanks.html. * 6. Stripe Dashboard > Developers > Webhooks > Add endpoint. Paste the same Web app URL. * Events to send: checkout.session.completed and checkout.session.expired. * * After editing this code later: Deploy > Manage deployments > edit the pencil icon > * Version: New version > Deploy. Otherwise the live site keeps running the old code. */ const SHEET_ID = '1-kSXVHEdHBnJad2cemaO4Qn_enqnVSYEAUPDCE2w2Cc'; const SITE_URL = 'https://maddieprole-coder.github.io/Honeymoon-Funds/'; const HOLD_MINUTES = 30; // how long a selection is held while the guest pays const EXPERIENCE_COUNT = 27; // rows 2-28 on the Experiences tab, in the same order as the website // ---------- Web app entry points ---------- function doGet(e) { const action = (e && e.parameter && e.parameter.action) || ''; if (action === 'progress') return json_({ funded: tally_().paid }); if (action === 'details') return json_(thankYouDetails_(e.parameter.res)); if (action === 'invite') return json_(inviteDetails_(e.parameter.code)); return json_({ ok: true }); } function doPost(e) { const body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); // Stripe webhook: Stripe's events have object === 'event'. Errors are thrown on purpose, // so Stripe retries the event later if something went wrong. if (body.object === 'event') { handleStripeEvent_(body.id); return json_({ received: true }); } // RSVP form (rsvp.html): save the reply to the RSVPs tab. if (body.type === 'rsvp') { try { return json_(saveRsvp_(body)); } catch (err) { console.error(err); return json_({ error: "We couldn't save your RSVP. Please try again." }); } } // Website request: reserve the selection and return a Stripe Checkout link. try { return json_(startCheckout_(body)); } catch (err) { console.error(err); return json_({ error: "We couldn't start the payment. Please try again." }); } } // ---------- One-off setup ---------- function setup() { const ss = SpreadsheetApp.openById(SHEET_ID); let alloc = ss.getSheetByName('Allocations'); if (!alloc) alloc = ss.insertSheet('Allocations'); alloc.getRange(1, 1, 1, 6).setValues([[ 'Reservation ID', 'Sheet row', 'Experience', 'Amount (£)', 'Status', 'Hold expires' ]]); alloc.setFrozenRows(1); // "Funded" on the Experiences tab is the total of paid amounts for that row. const exp = ss.getSheetByName('Experiences'); const formulas = []; for (let r = 2; r <= EXPERIENCE_COUNT + 1; r++) { formulas.push(['=SUMIFS(Allocations!$D:$D,Allocations!$B:$B,ROW(),Allocations!$E:$E,"Paid")']); } exp.getRange(2, 5, formulas.length, 1).setFormulas(formulas); } // ---------- Reserving a selection and starting Stripe Checkout ---------- function startCheckout_(body) { const name = String(body.name || '').trim().slice(0, 60); const message = String(body.message || '').trim().slice(0, 300); const wanted = Array.isArray(body.allocation) ? body.allocation : []; if (!name) return { error: 'Please add your name so we know who to thank.' }; if (!wanted.length) return { error: 'Please suggest some experiences first.' }; const ss = SpreadsheetApp.openById(SHEET_ID); // Columns A-D: name, category, price, quantity. The server uses these numbers, never the page's. const gifts = ss.getSheetByName('Experiences').getRange(2, 1, EXPERIENCE_COUNT, 4).getValues(); const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const { paid, held } = tally_(); const lines = []; let total = 0; for (const w of wanted) { const i = Number(w.exp); const amount = Math.round(Number(w.amount) * 100) / 100; if (!Number.isInteger(i) || i < 0 || i >= gifts.length || !(amount > 0)) { return { error: 'That selection did not work. Please click Suggest again.' }; } const expName = gifts[i][0]; const price = Number(gifts[i][2]); const qty = Number(gifts[i][3]); const left = price * qty - (paid[i] || 0) - (held[i] || 0); if (amount > left + 0.005) { return { error: '"' + expName + '" was just funded by someone else. Please click Shuffle for a new set.' }; } lines.push({ row: i + 2, name: expName, amount }); total += amount; } total = Math.round(total * 100) / 100; if (total < 1) return { error: 'The minimum contribution is £1.' }; const now = new Date(); const expires = new Date(now.getTime() + HOLD_MINUTES * 60000); const resId = 'R' + Utilities.formatDate(now, 'UTC', 'yyyyMMddHHmmss') + Math.floor(100 + Math.random() * 900); const summary = lines.map(l => l.name + ' (£' + l.amount + ')').join('; '); // Hold the amounts and record the request before going to Stripe. ss.getSheetByName('Reservations').appendRow([resId, now, name, message, summary, total, 'Pending']); const allocSheet = ss.getSheetByName('Allocations'); allocSheet.getRange(allocSheet.getLastRow() + 1, 1, lines.length, 6).setValues( lines.map(l => [resId, l.row, l.name, l.amount, 'Pending', expires]) ); // Optional tick box: covers the standard UK card fee (1.5% + 20p) on top of the gift. const coverFees = body.coverFees === true; const feeAmount = coverFees ? Math.round((total * 0.015 + 0.2) * 100) / 100 : 0; const feeLines = coverFees ? { 'line_items[1][price_data][currency]': 'gbp', 'line_items[1][price_data][unit_amount]': String(Math.round(feeAmount * 100)), 'line_items[1][price_data][product_data][name]': 'Towards card fees (optional)', 'line_items[1][quantity]': '1' } : {}; let session; try { session = stripe_('checkout/sessions', { mode: 'payment', 'line_items[0][price_data][currency]': 'gbp', 'line_items[0][price_data][unit_amount]': String(Math.round(total * 100)), 'line_items[0][price_data][product_data][name]': 'Honeymoon fund', 'line_items[0][price_data][product_data][description]': summary.slice(0, 450), 'line_items[0][quantity]': '1', ...feeLines, client_reference_id: resId, 'metadata[reservation_id]': resId, success_url: SITE_URL + 'thanks.html?res=' + resId, cancel_url: SITE_URL, // Stripe needs the session to last at least 30 minutes, so it runs just past the hold. expires_at: String(Math.floor(expires.getTime() / 1000) + 60) }); } catch (err) { console.error(err); setStatus_(resId, 'Failed'); return { error: "We couldn't start the payment. Please try again." }; } return { url: session.url }; } finally { lock.releaseLock(); } } // ---------- Stripe webhook ---------- function handleStripeEvent_(eventId) { // Fetch the event back from Stripe rather than trusting the request body. const event = stripe_('events/' + encodeURIComponent(eventId)); const session = event.data.object; const resId = session.metadata && session.metadata.reservation_id; if (!resId) return; const lock = LockService.getScriptLock(); lock.waitLock(30000); try { if (event.type === 'checkout.session.completed' && session.payment_status === 'paid') { markPaid_(resId, session); } else if (event.type === 'checkout.session.expired') { setStatus_(resId, 'Expired'); } } finally { lock.releaseLock(); } } function markPaid_(resId, session) { const ss = SpreadsheetApp.openById(SHEET_ID); const gifts = ss.getSheetByName('Gifts'); // Stripe can send the same event more than once, so skip any payment already recorded. const seen = gifts.getLastRow() > 1 ? gifts.getRange(2, 6, gifts.getLastRow() - 1, 1).getValues().flat() : []; if (seen.includes(session.id)) return; const res = ss.getSheetByName('Reservations'); const rows = res.getDataRange().getValues(); const r = rows.findIndex((row, idx) => idx > 0 && row[0] === resId); if (r < 0) throw new Error('Reservation not found: ' + resId); const row = rows[r]; setStatus_(resId, 'Paid'); gifts.appendRow([ Utilities.formatDate(new Date(), 'Europe/London', 'd MMM yyyy HH:mm'), row[2], // guest name row[3], // message row[4], // experiences chosen row[5], // gift amount in pounds (excludes the optional card-fee tick box) session.id, // Stripe checkout session ID 'Paid' ]); } function setStatus_(resId, status) { const ss = SpreadsheetApp.openById(SHEET_ID); [['Reservations', 7], ['Allocations', 5]].forEach(([tab, col]) => { const sh = ss.getSheetByName(tab); const rows = sh.getDataRange().getValues(); rows.forEach((row, r) => { if (r > 0 && row[0] === resId && row[col - 1] !== 'Paid') { sh.getRange(r + 1, col).setValue(status); } }); }); } // ---------- Helpers ---------- // Totals per experience index (0 = first experience on the website). // Paid counts toward progress. Pending holds count only until they expire. function tally_() { const paid = {}; const held = {}; const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName('Allocations'); const last = sh.getLastRow(); if (last < 2) return { paid, held }; const now = new Date(); sh.getRange(2, 1, last - 1, 6).getValues().forEach(row => { const i = Number(row[1]) - 2; const amount = Number(row[3]) || 0; if (row[4] === 'Paid') paid[i] = (paid[i] || 0) + amount; else if (row[4] === 'Pending' && new Date(row[5]) > now) held[i] = (held[i] || 0) + amount; }); return { paid, held }; } function stripe_(path, payload) { const key = PropertiesService.getScriptProperties().getProperty('STRIPE_SECRET_KEY'); if (!key) throw new Error('STRIPE_SECRET_KEY is not set in Script properties'); const opts = { method: payload ? 'post' : 'get', headers: { Authorization: 'Bearer ' + key }, muteHttpExceptions: true }; if (payload) { opts.contentType = 'application/x-www-form-urlencoded'; opts.payload = Object.entries(payload) .map(([k, v]) => encodeURIComponent(k) + '=' + encodeURIComponent(v)) .join('&'); } const res = UrlFetchApp.fetch('https://api.stripe.com/v1/' + path, opts); const data = JSON.parse(res.getContentText()); if (res.getResponseCode() >= 300) { throw new Error('Stripe error: ' + ((data.error && data.error.message) || res.getResponseCode())); } return data; } // ---------- Invitations (Guests tab) ---------- // Everyone on the invitation with this code. Guests tab columns: Code, Party, Type, Name, Menu, Link. function guestsForCode_(code) { const want = String(code || '').trim().toUpperCase(); if (!want) return []; const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName('Guests'); if (!sh || sh.getLastRow() < 2) return []; return sh.getRange(2, 1, sh.getLastRow() - 1, 5).getValues() .filter(r => String(r[0]).trim().toUpperCase() === want) .map(r => ({ party: String(r[1]), type: String(r[2]), name: String(r[3]), menu: String(r[4]) })); } // What the RSVP page needs to prefill an invitation. Unknown codes are refused. function inviteDetails_(code) { const people = guestsForCode_(code); if (!people.length) return { ok: false, error: "We couldn't find that invitation. Please check your link." }; return { ok: true, party: people[0].party, type: people[0].type, people: people.map(p => ({ name: p.name, menu: p.menu })) }; } // ---------- RSVPs ---------- // Adds one RSVP to the RSVPs tab, creating the tab with headings the first time. // Names come from the Guests tab, so guests can't change who the invitation is for. function saveRsvp_(body) { const invited = guestsForCode_(body.code); if (!invited.length) return { error: "This invitation link isn't recognised. Please check it and try again." }; const contact = invited[0].party; const clip = v => String(v || '').trim().slice(0, 300); const stamp = Utilities.formatDate(new Date(), 'Europe/London', 'd MMM yyyy HH:mm'); const notes = clip(body.notes); // One row per person on the invitation. A "no" gets a single row so it's still recorded. const code = String(body.code || '').trim().toUpperCase(); let rows; if (body.attending === 'yes') { const answers = Array.isArray(body.people) ? body.people : []; rows = invited.map((g, i) => { const a = answers[i] || {}; return [stamp, contact, 'Yes', clip(g.name || a.name), clip(a.starter), clip(a.main), clip(a.dessert), clip(a.allergies), clip(a.preferences), i === 0 ? notes : '', code]; }); } else { rows = [[stamp, contact, 'No', '', '', '', '', '', '', notes, code]]; } const ss = SpreadsheetApp.openById(SHEET_ID); let sh = ss.getSheetByName('RSVPs'); if (!sh) { sh = ss.insertSheet('RSVPs'); sh.appendRow(['Date', 'Contact', 'Attending', 'Guest name', 'Starter', 'Main', 'Dessert', 'Allergies', 'Preferences / intolerances', 'Notes', 'Code']); sh.setFrozenRows(1); } sh.getRange(sh.getLastRow() + 1, 1, rows.length, 11).setValues(rows); return { ok: true }; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Gift list for the thank-you page (thanks.html on GitHub). Experiences and wording only, never guest names. // Whole gifts read "You've allowed us to experience ...", part-contributions read "You've helped us get closer to experiencing ...". function thankYouDetails_(resId) { if (!resId) return { status: 'unknown', items: [] }; const ss = SpreadsheetApp.openById(SHEET_ID); const rows = ss.getSheetByName('Reservations').getDataRange().getValues(); const row = rows.find((r, idx) => idx > 0 && r[0] === resId); if (!row) return { status: 'unknown', items: [] }; const expRows = ss.getSheetByName('Experiences').getRange(2, 1, EXPERIENCE_COUNT, 3).getValues(); const priceOf = {}; expRows.forEach(r => { priceOf[r[0]] = Number(r[2]); }); const items = String(row[4]).split('; ').map(s => { const m = s.match(/^(.*) \(£([\d.]+)\)$/); if (!m) return s; const name = m[1]; const amt = Number(m[2]); const price = priceOf[name]; const whole = price > 0 && amt % price === 0; return whole ? "You've allowed us to experience " + name : "You've helped us get closer to experiencing " + name.replace(/^A share of /, ''); }); return { status: row[6] === 'Paid' ? 'paid' : 'pending', items }; }
